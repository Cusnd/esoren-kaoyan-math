\textbf{来源：} Codex 原创 / K064 深度讲义配套练习（非真题）

\begin{enumerate}
  \item[\textbf{A}] 设 \(f\in C[0,1]\cap D(0,1)\)，且
  \(f(0)=f(1)=0\)。证明：存在 \(\xi\in(0,1)\)，使
  \[
  f'(\xi)+2f(\xi)=0.
  \]

  \item[\textbf{B}] 设 \(f\in C[1,2]\cap D(1,2)\)，且
  \(f(2)=4f(1)\)。证明：存在 \(\xi\in(1,2)\)，使
  \[
  \xi f'(\xi)=2f(\xi).
  \]

  \item[\textbf{C}] 设 \(f\in C[0,1]\)，记
  \[
  I=\int_0^1 2x f(x)\,\mathrm dx.
  \]
  证明：存在 \(\xi\in(0,1)\)，使 \(f(\xi)=I\)。

  \item[\textbf{M}] 设 \(\lambda\in\mathbb R\)，
  \(f\in C[0,3]\)，且 \(f\) 在 \((0,3)\) 内二阶可导。已知
  \[
  f(0)=f(3)=0,\qquad f(1)=\lambda.
  \]
  证明：存在 \(\xi\in(0,3)\)，使
  \[
  f''(\xi)=-\lambda.
  \]
\end{enumerate}
\end{problemBox}

\begin{enumerate}
  \item[\textbf{A}] 判断下列极限能否直接使用洛必达法则，并求其极限
  （不存在时说明原因）：
  \[
  \begin{aligned}
  &(1)\ \lim_{x\to0}\frac{1-\cos x}{x^2},\qquad
  (2)\ \lim_{x\to0^+}x\ln x,\\
  &(3)\ \lim_{x\to0}\frac{1+x}{x},\qquad
  (4)\ \lim_{x\to+\infty}\frac{\ln x}{x}.
  \end{aligned}
  \]

  \item[\textbf{B}] 求极限
  \[
  \lim_{x\to0}\frac{\ln(1+x)-x}{x^2}.
  \]

  \item[\textbf{C}] 求极限
  \[
  \lim_{x\to0}\frac{\tan x-\sin x}{x^3}.
  \]

  \item[\textbf{M}] 对 \(x\ne0\)，令
  \[
  f(x)=x+x^2\sin\frac1{x^2},\qquad g(x)=x.
  \]
  证明
  \[
  \lim_{x\to0}\frac{f(x)}{g(x)}=1,
  \]
  但 \(\lim_{x\to0}f'(x)/g'(x)\) 不存在，并说明这反驳了洛必达法则的
  哪个错误逆命题。
\end{enumerate}
\end{problemBox}

\begin{enumerate}
  \item[\textbf{A1}] 从麦克劳林公式出发，将
  \(\sqrt{1+x}\) 展开到二阶，并写出皮亚诺余项。

  \item[\textbf{A2}] 将
  \[
  (1-2x)^{-1/2}
  \]
  展开到三阶，并写出皮亚诺余项。

  \item[\textbf{B1}] 求极限
  \[
  \lim_{x\to0}
  \frac{\sin x-x+\frac{x^3}{6}}{x^5}.
  \]

  \item[\textbf{B2}] 求极限
  \[
  \lim_{x\to0}\frac{\arctan x-\sin x}{x^3}.
  \]

  \item[\textbf{C}] 求极限
  \[
  \lim_{x\to0}
  \frac{\ln(1+x+x^2)-x-\frac{x^2}{2}}{x^3}.
  \]

  \item[\textbf{M}] 完成下列两项：
  \begin{enumerate}
    \item 求 \(a,b\)，使
    \[
    \lim_{x\to0}
    \frac{\ln(1+x)-x+ax^2+bx^3}{x^4}
    \]
    存在且有限，并求该极限；
    \item 用带拉格朗日余项的泰勒公式证明：当
    \(0\le x\le\pi/2\) 时，
    \[
    1-\frac{x^2}{2}
    \le \cos x
    \le 1-\frac{x^2}{2}+\frac{x^4}{24}.
    \]
  \end{enumerate}
\end{enumerate}
\end{problemBox}
