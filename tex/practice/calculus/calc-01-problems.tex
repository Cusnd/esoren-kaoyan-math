\textbf{OCR 校注：}第 1.14 题经放大核对，分母为 \(1+e^{4/x}\)。

\begin{enumerate}
  \item[\textbf{1.1}] 若
  \[
    \lim_{x\to0}\frac{\sin x+xf(x)}{x^3}=0,
  \]
  则 \(\displaystyle\lim_{x\to0}\frac{1+f(x)}{x^2}\) 为（\quad）。
  \[
    \textup{(A) }0\qquad
    \textup{(B) }\frac13\qquad
    \textup{(C) }\frac16\qquad
    \textup{(D) }\infty
  \]

  \item[\textbf{1.2}] 设 \(f(x)\) 在 \(( -\infty,+\infty )\) 内有定义，且
  \(\displaystyle\lim_{x\to+\infty}f(x)=a\)。令
  \[
    g(x)=
    \begin{cases}
      f\!\left(\dfrac1x\right),&x\ne0,\\
      0,&x=0,
    \end{cases}
  \]
  则（\quad）。
  \begin{enumerate}
    \item[\textup{(A)}] \(x=0\) 必是 \(g(x)\) 的第一类间断点；
    \item[\textup{(B)}] \(x=0\) 必是 \(g(x)\) 的第二类间断点；
    \item[\textup{(C)}] \(x=0\) 必是 \(g(x)\) 的连续点；
    \item[\textup{(D)}] \(g(x)\) 在 \(x=0\) 处的连续性与 \(a\) 的取值有关。
  \end{enumerate}

  \item[\textbf{1.3}] 设
  \[
    f(x)=\frac1{e^{x/(x-1)}-1},
  \]
  判断 \(x=0\) 与 \(x=1\) 的间断点类型。

  \item[\textbf{1.4}] 函数
  \[
    f(x)=\lim_{t\to0}\left(1+\frac{\sin t}{x}\right)^{x^2/t}
  \]
  在 \(( -\infty,+\infty )\) 内（\quad）。
  \[
    \textup{(A) 连续}\qquad
    \textup{(B) 有可去间断点}\qquad
    \textup{(C) 有跳跃间断点}\qquad
    \textup{(D) 有无穷间断点}
  \]

  \item[\textbf{1.5}] 设
  \[
    f(x)=\lim_{n\to\infty}\frac{1+x}{1+x^{2n}},
  \]
  判断该函数的间断点。

  \item[\textbf{1.6}] 设 \(f(x)=\dfrac{x}{1+x}\)，\(x\in[0,1]\)，定义
  \[
    f_1(x)=f(x),\qquad f_n(x)=f\!\left[f_{n-1}(x)\right]\quad(n\ge2),
  \]
  求 \(f_n(x)\)。

  \item[\textbf{1.7}] 求
  \[
    \lim_{x\to0^+}\frac{1-\sqrt{\cos x}}{x\left(1-\cos\sqrt{x}\right)}.
  \]

  \item[\textbf{1.8}] 求
  \[
    \lim_{x\to\infty}\left(\frac{x+2}{x-1}\right)^x.
  \]

  \item[\textbf{1.9}] 已知 \(a>0,b>0\)，求
  \[
    \lim_{x\to+\infty}x\left(a^{1/x}-b^{1/x}\right).
  \]

  \item[\textbf{1.10}] 设
  \[
    f(x)=
    \begin{cases}
      \dfrac{1-e^{\tan x}}{\arcsin(x/2)},&x>0,\\[6pt]
      ae^{2x},&x\le0.
    \end{cases}
  \]
  若 \(f(x)\) 在 \(x=0\) 处连续，求 \(a\)。

  \item[\textbf{1.11}] 设 \(f(x)=\arcsin(\sin x)\)，画出 \(f(x)\) 的图形。

  \item[\textbf{1.12}] 求
  \[
    \lim_{x\to0}\left(\frac{e^x+xe^x}{e^x-1}-\frac1x\right).
  \]

  \item[\textbf{1.13}] 已知 \(a_i>0\ (i=1,2,\ldots,n)\)，求
  \[
    \lim_{x\to0}
    \left(\frac{a_1^x+a_2^x+\cdots+a_n^x}{n}\right)^{n/x}.
  \]

  \item[\textbf{1.14}] 已知
  \[
    \lim_{x\to0}\left[
      a\frac{2+e^{1/x}}{1+e^{4/x}}+(1+|x|)^{1/x}
    \right]
  \]
  存在，求 \(a\)。

  \item[\textbf{1.15}] 设
  \[
    \lim_{x\to0}\frac{\ln(1+x)-(ax+bx^2)}{x^2}=2,
  \]
  求常数 \(a,b\)。

  \item[\textbf{1.16}] 求常数 \(a,b\)，使
  \[
    f(x)=x-(a+b\cos x)\sin x
  \]
  在 \(x\to0\) 时是 \(x\) 的五阶无穷小。
\end{enumerate}
\end{problemBox}
